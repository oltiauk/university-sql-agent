# M1 Database Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A seeded university SQLite database and a small read-only DB layer the agent will use in M2.

**Architecture:** Two SQL files define and fill the database; a build script turns them into a `.db` file. `agent/db.py` opens any SQLite file read-only, describes it as text, and runs one guarded `SELECT`. Nothing in `agent/` knows what the tables mean.

**Tech Stack:** Python 3.11+, stdlib `sqlite3`, `pytest`. No other dependencies in this milestone.

**Spec:** `docs/superpowers/specs/2026-11-25-university-sql-agent-design.md` (sections Database, DB layer, Tests, M1)

## Global Constraints

- Python 3.11+. Only `pytest` is installed in this milestone.
- `agent/` must not contain the words student, teacher, course, enrollment, grade, semester (any case), in code, comments or strings. A test enforces this in M2; write to it now.
- Semester values are written `'2025-Fall'` and `'2026-Spring'`.
- Student names are single first names and unique.
- The DB layer opens files with `file:<path>?mode=ro`.
- The guard accepts exactly one statement starting with `SELECT` or `WITH`.
- Query results are capped at 100 rows with a `truncated` flag.
- `get_schema` returns each table's `CREATE TABLE` text followed by its first 3 rows.
- Mark any deliberate shortcut with a `# ponytail:` comment naming its ceiling.
- No file under `docs/` is changed by this plan.

## Review Focus

1. Model output often ends with `;` or uses lowercase: `"select 1;"` and `"  SELECT 1  "` must run, not be rejected. (Task 2, `test_guard_accepts`)
2. Model output wrapped in a code fence or starting with a comment must be rejected with a `ValueError`, not crash or run. (Task 2, `test_guard_rejects`)
3. A database path that does not exist must raise an error and must not create an empty file. (Task 2, `test_connect_missing_file`)
4. A table with no rows, or a name that needs quoting, must still appear in the schema text. (Task 2, `test_schema_odd_tables`)
5. Exactly 100 rows is not truncated; 101 is. (Task 2, `test_row_cap`)
6. Building twice must give the same database, not doubled rows. (Task 1, `test_build_twice`)

CLASS = """FROM enrollments e
JOIN students s ON s.id = e.student_id
JOIN offerings o ON o.id = e.offering_id
JOIN courses c ON c.id = o.course_id
WHERE c.title = 'Databases' AND o.semester = '2025-Fall'"""

def test_five_tables(db_path):
    names = {r[0] for r in rows(db_path, "SELECT name FROM sqlite_master WHERE type='table'")}
    assert names == {"teachers", "students", "courses", "offerings", "enrollments"}

def test_databases_fall_class(db_path):
    got = rows(db_path, f"SELECT s.name, e.grade {CLASS} ORDER BY e.grade DESC")
    assert got == [("Arta", 95), ("Blerim", 88), ("Dren", 82), ("Era", 70), ("Fatos", 55)]

def test_above_class_average(db_path):
    got = rows(db_path, f"""SELECT s.name {CLASS} AND e.grade > (
        SELECT AVG(e2.grade) FROM enrollments e2 WHERE e2.offering_id = o.id) ORDER BY s.name""")
    assert got == [("Arta",), ("Blerim",), ("Dren",)]

def test_overall_average_gives_a_different_answer(db_path):
    assert rows(db_path, "SELECT AVG(grade) FROM enrollments") == [(87.0,)]
    got = rows(db_path, f"SELECT s.name {CLASS} AND e.grade > (SELECT AVG(grade) FROM enrollments) ORDER BY s.name")
    assert got == [("Arta",), ("Blerim",)]

def test_people_and_enrollments_differ(db_path):
    got = rows(db_path, """SELECT COUNT(*), COUNT(DISTINCT e.student_id) FROM enrollments e
        JOIN offerings o ON o.id = e.offering_id JOIN teachers t ON t.id = o.teacher_id
        WHERE t.name = 'Besnik Krasniqi'""")
    assert got == [(6, 5)]

def test_average_by_teacher(db_path):
    got = dict(rows(db_path, """SELECT t.name, ROUND(AVG(e.grade), 1) FROM enrollments e
        JOIN offerings o ON o.id = e.offering_id JOIN teachers t ON t.id = o.teacher_id GROUP BY t.id"""))
    assert got == {"Lira Hoxha": 83.0, "Besnik Krasniqi": 91.3, "Mira Berisha": 89.0}

def test_semester_filter(db_path):
    got = rows(db_path, """SELECT COUNT(*) FROM enrollments e JOIN offerings o ON o.id = e.offering_id
        WHERE o.semester = '2026-Spring'""")
    assert got == [(7,)]

def test_totals(db_path):
    assert rows(db_path, "SELECT COUNT(*) FROM enrollments") == [(19,)]
    assert rows(db_path, "SELECT COUNT(*) FROM enrollments WHERE grade IS NULL") == [(2,)]
    assert rows(db_path, "SELECT COUNT(*), COUNT(DISTINCT name) FROM students") == [(8, 8)]

def test_constraints(db_path):
    conn = sqlite3.connect(db_path)
    for bad in ("INSERT INTO students(name) VALUES (NULL)",
                "INSERT INTO teachers(name) VALUES (NULL)",
                "INSERT INTO enrollments(student_id, offering_id, grade) VALUES (1, 6, 101)",
                "INSERT INTO enrollments(student_id, offering_id, grade) VALUES (1, 1, 50)"):
        with pytest.raises(sqlite3.IntegrityError):
            conn.execute(bad)
    assert conn.execute("PRAGMA foreign_key_check").fetchall() == []

def test_build_twice(db_path):
    build(ROOT / "db/schema.sql", ROOT / "db/seed.sql", db_path)
    assert rows(db_path, "SELECT COUNT(*) FROM students") == [(8,)]

- [ ] **Step 4: Write `db/schema.sql`**

```sql
CREATE TABLE teachers (
    id   INTEGER PRIMARY KEY,
    name TEXT NOT NULL
);

CREATE TABLE students (
    id   INTEGER PRIMARY KEY,
    name TEXT NOT NULL
);

CREATE TABLE courses (
    id    INTEGER PRIMARY KEY,
    code  TEXT NOT NULL UNIQUE,
    title TEXT NOT NULL
);

-- One course taught by one teacher in one semester, e.g. '2025-Fall'.
CREATE TABLE offerings (
    id         INTEGER PRIMARY KEY,
    course_id  INTEGER NOT NULL REFERENCES courses(id),
    teacher_id INTEGER NOT NULL REFERENCES teachers(id),
    semester   TEXT NOT NULL,
    UNIQUE (course_id, teacher_id, semester)
);

-- A student in an offering. grade is 0 to 100, NULL until graded.
CREATE TABLE enrollments (
    student_id  INTEGER NOT NULL REFERENCES students(id),
    offering_id INTEGER NOT NULL REFERENCES offerings(id),
    grade       REAL CHECK (grade BETWEEN 0 AND 100),
    PRIMARY KEY (student_id, offering_id)
);
```

### Task 2: Read-only DB layer

**Files:**
- Create: `agent/__init__.py` (empty), `agent/db.py`
- Test: `tests/test_db.py`

**Interfaces:**
- Consumes: fixture `db_path` from Task 1.
- Produces, all in `agent/db.py`:
  - `connect(db_path: str) -> sqlite3.Connection`: read-only connection.
  - `get_schema(conn: sqlite3.Connection) -> str`: for every table in `sqlite_master` (skipping names starting with `sqlite_`), the `CREATE TABLE` text, then the column names and up to 3 rows from `SELECT * FROM "<name>" LIMIT 3`. Tables separated by a blank line.
  - `check_sql(sql: str) -> None`: raises `ValueError` with a message a model can act on.
  - `run_query(conn: sqlite3.Connection, sql: str, max_rows: int = 100) -> tuple[list[str], list[tuple], bool]`: calls `check_sql`, runs the query, returns `(columns, rows, truncated)`. `ValueError` and `sqlite3.Error` propagate to the caller.

- [ ] **Step 1: Write the failing tests** in `tests/test_db.py`

```python
from agent.db import check_sql, connect, get_schema, run_query

def test_connection_is_read_only(db_path):
    conn = connect(db_path)
    with pytest.raises(sqlite3.OperationalError):
        conn.execute("DELETE FROM students")
    with pytest.raises(sqlite3.OperationalError):
        conn.execute("WITH x AS (SELECT 1) DELETE FROM students")

def test_connect_missing_file(tmp_path):
    missing = tmp_path / "nope.db"
    with pytest.raises(sqlite3.OperationalError):
        connect(str(missing)).execute("SELECT 1")
    assert not missing.exists()

def test_schema_lists_tables_and_sample_rows(db_path):
    schema = get_schema(connect(db_path))
    for name in ("teachers", "students", "courses", "offerings", "enrollments"):
        assert f"CREATE TABLE {name}" in schema
    assert "2025-Fall" in schema          # value format visible to the model
    assert "Arta" in schema and "Era" not in schema   # first 3 rows only
    assert "sqlite_" not in schema

def test_schema_odd_tables(tmp_path):
    path = tmp_path / "odd.db"
    rw = sqlite3.connect(path)
    rw.executescript('CREATE TABLE empty_one (a TEXT); CREATE TABLE "order" (x INT); INSERT INTO "order" VALUES (7);')
    rw.close()
    schema = get_schema(connect(str(path)))
    assert "CREATE TABLE empty_one" in schema and 'CREATE TABLE "order"' in schema and "7" in schema

@pytest.mark.parametrize("sql", ["SELECT 1", "select 1;", "  SELECT 1  \n", "WITH x AS (SELECT 1) SELECT * FROM x"])
def test_guard_accepts(sql):
    check_sql(sql)

@pytest.mark.parametrize("sql", [
    "", "INSERT INTO t VALUES (1)", "DROP TABLE t", "UPDATE t SET a = 1", "PRAGMA table_info(t)",
    "SELECT 1; SELECT 2", "SELECT 1; DROP TABLE t",
    "```sql\nSELECT 1\n```", "-- note\nSELECT 1",
])
def test_guard_rejects(sql):
    with pytest.raises(ValueError):
        check_sql(sql)

def test_run_query(db_path):
    cols, rows, truncated = run_query(connect(db_path), "SELECT id, name FROM students ORDER BY id LIMIT 2;")
    assert (cols, rows, truncated) == (["id", "name"], [(1, "Arta"), (2, "Blerim")], False)

def test_run_query_guards_first(db_path):
    with pytest.raises(ValueError):
        run_query(connect(db_path), "DELETE FROM students")

def test_run_query_bad_sql_raises_db_error(db_path):
    with pytest.raises(sqlite3.Error):
        run_query(connect(db_path), "SELECT nope FROM students")

def test_row_cap(tmp_path):
    path = tmp_path / "many.db"
    rw = sqlite3.connect(path)
    rw.execute("CREATE TABLE n (i INT)")
    rw.executemany("INSERT INTO n VALUES (?)", [(i,) for i in range(101)])
    rw.commit(); rw.close()
    conn = connect(str(path))
    _, rows, truncated = run_query(conn, "SELECT i FROM n WHERE i < 100")
    assert (len(rows), truncated) == (100, False)
    _, rows, truncated = run_query(conn, "SELECT i FROM n")
    assert (len(rows), truncated) == (100, True)
```

- [ ] **Step 3: Implement the four functions in `agent/db.py`**

- `connect`: `sqlite3.connect(f"file:{db_path}?mode=ro", uri=True)`.
- `check_sql`: strip whitespace and one trailing `;`. Reject if empty, if a `;` remains, or if the first word (case-insensitive) is not `SELECT` or `WITH`. Rejecting any remaining `;` also rejects a semicolon inside a quoted string; mark it `# ponytail:` with that ceiling. The messages: "Only a single SELECT statement is allowed." Writes hidden behind `WITH` are stopped by the read-only connection, not by this function.
- `run_query`: `fetchmany(max_rows + 1)`; `truncated` is whether more than `max_rows` came back; return the first `max_rows`. Columns come from `cursor.description`.
- `get_schema`: quote the table name with double quotes in the sample query.
